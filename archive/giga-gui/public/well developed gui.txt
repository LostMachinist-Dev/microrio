/* ============================================================
   CONTROL BUTTON HANDLERS
   ============================================================ */

function ctrlInc(i) {
    const valEl = document.getElementById(`ctrl_dbl_val_${i}`);

    let inc = parseFloat(document.getElementById(`ctrl_inc_${i}`).value);
    let max = parseFloat(document.getElementById(`ctrl_max_${i}`).value);
    let v   = parseFloat(valEl.value);

    if (isNaN(inc)) inc = 1;
    if (isNaN(max)) max = Infinity;
    if (isNaN(v))   v = 0;

    v = Math.min(v + inc, max);
    valEl.value = v;
}

function ctrlDec(i) {
    const valEl = document.getElementById(`ctrl_dbl_val_${i}`);

    let inc = parseFloat(document.getElementById(`ctrl_inc_${i}`).value);
    let min = parseFloat(document.getElementById(`ctrl_min_${i}`).value);
    let v   = parseFloat(valEl.value);

    if (isNaN(inc)) inc = 1;
    if (isNaN(min)) min = -Infinity;
    if (isNaN(v))   v = 0;

    v = Math.max(v - inc, min);
    valEl.value = v;
}

function toggleCtrlBool(i) {
    const el = document.getElementById(`ctrl_bool_${i}`);
    const isOn = el.classList.contains("led-on");

    el.classList.toggle("led-on", !isOn);
    el.classList.toggle("led-off", isOn);
}


/* ============================================================
   WEBSOCKET TELEMETRY UPDATE (INBOUND)
   ============================================================ */

const ws = new WebSocket("ws://localhost:8080");

ws.onmessage = (event) => {
    const msg = JSON.parse(event.data);

    if (msg.type !== "arduino") return;

    const data = msg.data;

    data.bools.forEach((v, i) => {
        const cell = document.getElementById(`bool_cell_${i}`);
        if (!cell) return;
        cell.classList.toggle("bool-high", v === true || v === 1);
    });

    data.doubles.forEach((v, i) => {
        const rawEl = document.getElementById(`dbl_raw_${i}`);
        const scaledEl = document.getElementById(`dbl_scaled_${i}`);
        const slope = parseFloat(document.getElementById(`slope_${i}`).value) || 0;
        const offset = parseFloat(document.getElementById(`offset_${i}`).value) || 0;

        rawEl.textContent = v.toFixed(3);
        scaledEl.textContent = (v * slope + offset).toFixed(3);
    });

    data.strings.forEach((v, i) => {
        const el = document.getElementById(`str_val_${i}`);
        if (el) el.textContent = v;
    });
};


/* ============================================================
   GAMEPAD NAME → INPUT BINDING MAP
   ============================================================ */

const gpMap = {};
for (let a = 0; a < 16; a++) gpMap[`GPA${a}`] = { type: "axis", index: a };
for (let b = 0; b < 16; b++) gpMap[`GPB${b}`] = { type: "button", index: b };


/* ============================================================
   APPLY JOYSTICK VALUES TO CONTROLS
   ============================================================ */

function applyJoystickToControls(joy) {
    if (!joy) return;

    for (let i = 0; i < 32; i++) {

        const dblName = document.getElementById(`ctrl_dbl_name_${i}`).value.trim();
        const dblMap = gpMap[dblName];

        if (dblMap && dblMap.type === "axis") {
            let raw = joy.axes[dblMap.index] ?? 0;

            const min = parseFloat(document.getElementById(`ctrl_min_${i}`).value);
            const max = parseFloat(document.getElementById(`ctrl_max_${i}`).value);

            if (!isNaN(min)) raw = Math.max(raw, min);
            if (!isNaN(max)) raw = Math.min(raw, max);

            document.getElementById(`ctrl_dbl_val_${i}`).value = raw;
        }

        const boolName = document.getElementById(`ctrl_bool_name_${i}`).value.trim();
        const boolMap = gpMap[boolName];

        if (boolMap && boolMap.type === "button") {
            const pressed = joy.buttons[boolMap.index]?.pressed ?? false;
            const el = document.getElementById(`ctrl_bool_${i}`);

            el.classList.toggle("led-on", pressed);
            el.classList.toggle("led-off", !pressed);
        }
    }
}


/* ============================================================
   BROWSER GAMEPAD POLLING (30 ms)
   ============================================================ */

function pollGamepad() {
    const gp = navigator.getGamepads()[0];
    if (gp) applyJoystickToControls(gp);
}
setInterval(pollGamepad, 30);


/* ============================================================
   OUTBOUND CONTROL PACKET (GUI → Node)
   ============================================================ */

function collectOutputs() {
    const out = {
        c_bools:   new Array(32),
        c_doubles: new Array(32),
        c_strings: new Array(32)
    };

    for (let i = 0; i < 32; i++) {

        out.c_bools[i] =
            document.getElementById(`ctrl_bool_${i}`).classList.contains("led-on") ? 1 : 0;

        const v = parseFloat(document.getElementById(`ctrl_dbl_val_${i}`).value);
        out.c_doubles[i] = isNaN(v) ? 0 : v;

        out.c_strings[i] =
            document.getElementById(`ctrl_str_val_${i}`).value;
    }

    return out;
}


/* ============================================================
   FIXED-RATE OUTBOUND CONTROL LOOP (20 Hz)
   ============================================================ */

function sendControlFrame() {
    const packet = {
        type: "controls",
        data: collectOutputs()
    };
    ws.send(JSON.stringify(packet));
}

setInterval(sendControlFrame, 50); // 20 Hz
