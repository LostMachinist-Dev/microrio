            const DOUBLE_COUNT = 16;

            const INT_COUNT = 16;

function createNumericRow(index, panelElement, arrays) {
    const row = document.createElement("div");
    row.className = "numeric-row";

    // --- Chart enable LED ---
    const chartLed = document.createElement("div");
    chartLed.className = "chart-led";
    chartLed.dataset.enabled = "0"; // off by default
    chartLed.onclick = () => {
        const enabled = chartLed.dataset.enabled === "1";
        chartLed.dataset.enabled = enabled ? "0" : "1";
        chartLed.classList.toggle("on", !enabled);
        updateActiveChannelList();
    };

    // --- Axis selector ---
    const axisBtn = document.createElement("button");
    axisBtn.className = "axis-btn";
    axisBtn.textContent = "L";
    axisBtn.dataset.axis = "L";
    axisBtn.onclick = () => {
        axisBtn.dataset.axis = axisBtn.dataset.axis === "L" ? "R" : "L";
        axisBtn.textContent = axisBtn.dataset.axis;
        updateActiveChannelList();
    };

    // --- Name ---
    const name = document.createElement("input");
    name.className = "numeric-name";
    name.type = "text";
    name.value = `${arrays.labelPrefix} ${index}`;

    // --- Raw ---
    const raw = document.createElement("input");
    raw.className = "numeric-raw";
    raw.type = "text";
    raw.readOnly = true;

    // --- EU ---
    const eu = document.createElement("input");
    eu.className = "numeric-eu";
    eu.type = "text";
    eu.readOnly = true;

    // --- Slope ---
    const slope = document.createElement("input");
    slope.className = "numeric-slope";
    slope.type = "number";
    slope.value = "1.0";

    // --- Offset ---
    const offset = document.createElement("input");
    offset.className = "numeric-offset";
    offset.type = "number";
    offset.value = "0.0";

    // Append in order
    row.appendChild(chartLed);
    row.appendChild(axisBtn);
    row.appendChild(name);
    row.appendChild(raw);
    row.appendChild(eu);
    row.appendChild(slope);
    row.appendChild(offset);

    panelElement.appendChild(row);

    // Store references
    arrays.chartLeds.push(chartLed);
    arrays.axisBtns.push(axisBtn);
    arrays.names.push(name);
    arrays.raws.push(raw);
    arrays.eus.push(eu);
    arrays.slopes.push(slope);
    arrays.offsets.push(offset);
}

            // Storage arrays for each datatype
const doubles = { names: [], raws: [], eus: [], slopes: [], offsets: [], chartLeds: [], axisBtns: [], labelPrefix: "Double" };
const floats  = { names: [], raws: [], eus: [], slopes: [], offsets: [], chartLeds: [], axisBtns: [], labelPrefix: "Float" };
const ints    = { names: [], raws: [], eus: [], slopes: [], offsets: [], chartLeds: [], axisBtns: [], labelPrefix: "Int" };

            // Build doubles
            const doublePanel = document.getElementById("doublePanel");
            for (let i = 0; i < DOUBLE_COUNT; i++) {
                createNumericRow(i, doublePanel, doubles);
            }

            // Build floats
            const floatPanel = document.getElementById("floatPanel");
            for (let i = 0; i < FLOAT_COUNT; i++) {
                createNumericRow(i, floatPanel, floats);
            }

            // Build ints
            const intPanel = document.getElementById("intPanel");
            for (let i = 0; i < INT_COUNT; i++) {
                createNumericRow(i, intPanel, ints);
            }


        /* ------------------------------------------------------------
           HANDLE INCOMING DATA FROM ARDUINO
           ------------------------------------------------------------ */
        ws.onmessage = (event) => {
            const msg = JSON.parse(event.data);

            if (msg.type === "arduino") {

                /* Update bool LEDs */
                const bools = msg.data.bools;
                for (let i = 0; i < 32; i++) {
                    if (bools[i]) {
                        boolElements[i].classList.add("on");
                    } else {
                        boolElements[i].classList.remove("on");
                    }
                }

                /* Update strings */
                if (msg.data.strings) {
                    msg.data.strings.forEach((str, i) => {
                        if (stringValues[i]) {
                            stringValues[i].value = str;
                        }
                    });
                }
            }
            // ---------- DOUBLES ----------
                if (msg.data.doubles) {
                    msg.data.doubles.forEach((value, i) => {
                        if (doubles.raws[i]) {
                            doubles.raws[i].value = value;
                            const slope = parseFloat(doubles.slopes[i].value);
                            const offset = parseFloat(doubles.offsets[i].value);
                            doubles.eus[i].value = (value * slope + offset).toFixed(3);
                        }
                    });
                }

                // ---------- FLOATS ----------
                if (msg.data.floats) {
                    msg.data.floats.forEach((value, i) => {
                        if (floats.raws[i]) {
                            floats.raws[i].value = value;
                            const slope = parseFloat(floats.slopes[i].value);
                            const offset = parseFloat(floats.offsets[i].value);
                            floats.eus[i].value = (value * slope + offset).toFixed(3);
                        }
                    });
                }

                // ---------- INTS ----------
                if (msg.data.ints) {
                    msg.data.ints.forEach((value, i) => {
                        if (ints.raws[i]) {
                            ints.raws[i].value = value;
                            const slope = parseFloat(ints.slopes[i].value);
                            const offset = parseFloat(ints.offsets[i].value);
                            ints.eus[i].value = (value * slope + offset).toFixed(3);
                        }
                    });
                }

        };

document.getElementById("loadFile").onchange = (event) => {
    const file = event.target.files[0];
    if (!file) return;

    const reader = new FileReader();
    reader.onload = (e) => {
        const data = JSON.parse(e.target.result);

        // ---------- LOAD NAMES ----------
        if (data.boolNames) {
            data.boolNames.forEach((v, i) => nameInputs[i] && (nameInputs[i].value = v));
        }
        if (data.stringNames) {
            data.stringNames.forEach((v, i) => stringLabels[i] && (stringLabels[i].value = v));
        }
        if (data.doubleNames) {
            data.doubleNames.forEach((v, i) => doubles.names[i] && (doubles.names[i].value = v));
        }
        if (data.floatNames) {
            data.floatNames.forEach((v, i) => floats.names[i] && (floats.names[i].value = v));
        }
        if (data.intNames) {
            data.intNames.forEach((v, i) => ints.names[i] && (ints.names[i].value = v));
        }

        // ---------- LOAD OFFSETS ----------
        if (data.doubleSlopes) {
            data.doubleSlopes.forEach((v, i) => doubles.slopes[i] && (doubles.slopes[i].value = v));
        }
        if (data.doubleOffsets) {
            data.doubleOffsets.forEach((v, i) => doubles.offsets[i] && (doubles.offsets[i].value = v));
        }

        if (data.floatSlopes) {
            data.floatSlopes.forEach((v, i) => floats.slopes[i] && (floats.slopes[i].value = v));
        }
        if (data.floatOffsets) {
            data.floatOffsets.forEach((v, i) => floats.offsets[i] && (floats.offsets[i].value = v));
        }

        if (data.intSlopes) {
            data.intSlopes.forEach((v, i) => ints.slopes[i] && (ints.slopes[i].value = v));
        }
        if (data.intOffsets) {
            data.intOffsets.forEach((v, i) => ints.offsets[i] && (ints.offsets[i].value = v));
        }
    };

    reader.readAsText(file);
};
