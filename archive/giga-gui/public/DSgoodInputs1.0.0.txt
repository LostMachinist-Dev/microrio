<script>
/* ============================================================
   BUILD GRID ROWS
   ============================================================ */

const left = document.getElementById("leftPanel");

// Clear the example row
left.innerHTML = "";

for (let i = 0; i < 32; i++) {

    // Bool name (separator on the right)
    left.innerHTML += `
        <div class="cell sep-right" id="bool_cell_${i}">
            <input id="bool_name_${i}" value="B${i}">
        </div>`;

    // Double name (muted)
    left.innerHTML += `
        <div class="cell input-muted">
            <input class="input-muted" id="dbl_name_${i}" value="D${i}">
        </div>`;

    // Double raw (muted)
    left.innerHTML += `
        <div class="cell cell-muted" id="dbl_raw_${i}">0</div>`;

    // Double scaled (normal)
    left.innerHTML += `
        <div class="cell" id="dbl_scaled_${i}">0</div>`;

    // Slope (muted)
    left.innerHTML += `
        <div class="cell input-muted">
            <input class="input-muted" id="slope_${i}" value="1">
        </div>`;

    // Offset (muted + separator on the right)
    left.innerHTML += `
        <div class="cell input-muted sep-right">
            <input class="input-muted" id="offset_${i}" value="0">
        </div>`;

    // String name (muted)
    left.innerHTML += `
        <div class="cell input-muted">
            <input class="input-muted" id="str_name_${i}" value="S${i}">
        </div>`;

    // String value (normal)
    left.innerHTML += `
        <div class="cell" id="str_val_${i}">-</div>`;
}


/* ============================================================
   WEBSOCKET TELEMETRY UPDATE
   ============================================================ */

const ws = new WebSocket("ws://localhost:8080");

ws.onmessage = (event) => {
    const msg = JSON.parse(event.data);
    if (msg.type !== "arduino") return;

    const data = msg.data;

    // Update bools (LED highlight)
    data.bools.forEach((v, i) => {
        const cell = document.getElementById(`bool_cell_${i}`);
        if (!cell) return;
        cell.classList.toggle("bool-high", v === true || v === 1);
    });

    // Update doubles
    data.doubles.forEach((v, i) => {
        const rawEl = document.getElementById(`dbl_raw_${i}`);
        const scaledEl = document.getElementById(`dbl_scaled_${i}`);
        const slope = parseFloat(document.getElementById(`slope_${i}`).value) || 0;
        const offset = parseFloat(document.getElementById(`offset_${i}`).value) || 0;

        rawEl.textContent = v.toFixed(3);
        scaledEl.textContent = (v * slope + offset).toFixed(3);
    });

    // Update strings
    data.strings.forEach((v, i) => {
        const el = document.getElementById(`str_val_${i}`);
        if (el) el.textContent = v;
    });
};
